**PreviewPanel.h:**
```cpp
#pragma once
#include <windows.h>

class PreviewPanel {
public:
    void init(HINSTANCE hInst, HWND nppHandle);
    void destroy();
    void toggle(int cmdID);  // cmdID for checkmark
    bool isVisible() const;
    HWND getHwnd() const;

private:
    void registerPanel();
    void createHostWindow();
    static LRESULT CALLBACK wndProc(HWND hWnd, UINT msg, WPARAM wParam, LPARAM lParam);

    HINSTANCE m_hInst = nullptr;
    HWND m_nppHandle = nullptr;
    HWND m_hPanel = nullptr;  // The host window registered as docking panel
    int m_cmdID = 0;
    bool m_isRegistered = false;
    bool m_isVisible = false;
};
```

**PreviewPanel.cpp:**

1. `init(hInst, nppHandle)`: Store handles. Do NOT create the window yet (lazy per D-06 — window created on first toggle).

2. `createHostWindow()`: Register a Win32 window class "MarkdownPreviewPanel" and create a child window:
   - Style: WS_CHILD | WS_CLIPCHILDREN
   - Parent: m_nppHandle
   - Width: 50% of Notepad++ client area width (per D-03). Get via `GetClientRect(m_nppHandle, &rect)`, then use `rect.right / 2` as width.
   - Height: full client height
   - Store `this` pointer via SetWindowLongPtr(GWLP_USERDATA) for WndProc dispatch

3. `registerPanel()`: Called once on first toggle. Creates the host window, then:
   ```cpp
   tTbData dockData = {};
   dockData.hClient = m_hPanel;
   dockData.pszName = L"Markdown Preview";  // Panel title
   dockData.dlgID = 0;  // Index into funcItems
   dockData.uMask = DWS_DF_CONT_RIGHT;  // Per D-01: dock right by default
   dockData.pszModuleName = L"MarkdownPreview.dll";
   ::SendMessage(m_nppHandle, NPPM_DMMREGASDCKDLG, 0, (LPARAM)&dockData);
   ```
   Note per Pitfall 4: register ONCE, then only use DMMSHOW/DMMHIDE. D-02 (redocking) is handled automatically by Notepad++ after registration.

4. `toggle(cmdID)`:
   ```cpp
   m_cmdID = cmdID;
   if (!m_isRegistered) {
       registerPanel();
       m_isRegistered = true;
   }
   if (m_isVisible) {
       ::SendMessage(m_nppHandle, NPPM_DMMHIDE, 0, (LPARAM)m_hPanel);
       m_isVisible = false;
   } else {
       ::SendMessage(m_nppHandle, NPPM_DMMSHOW, 0, (LPARAM)m_hPanel);
       m_isVisible = true;
   }
   // Update menu checkmark per UI-SPEC Interaction Contract
   ::SendMessage(m_nppHandle, NPPM_SETMENUITEMCHECK, m_cmdID, m_isVisible);
   ```

5. `wndProc`: Handle WM_SIZE (will be needed for WebView2 resize in Plan 03). For now, just `DefWindowProc`.

6. `destroy()`: DestroyWindow if created.

7. `isVisible()`: return m_isVisible.
8. `getHwnd()`: return m_hPanel.

**Settings.h:**

```cpp
#pragma once

struct Settings {
    bool panelVisible = false;

    void load(const std::wstring& configPath);
    void save(const std::wstring& configPath);
};
```

**Settings.cpp:**
```cpp
#include "Settings.h"
#include <nlohmann/json.hpp>
#include <fstream>

void Settings::load(const std::wstring& configPath) {
    std::ifstream f(configPath);
    if (!f.is_open()) return;
    try {
        nlohmann::json j = nlohmann::json::parse(f);
        panelVisible = j.value("panelVisible", false);
    } catch (...) {
        // Use defaults on parse error
    }
}

void Settings::save(const std::wstring& configPath) {
    nlohmann::json j;
    j["panelVisible"] = panelVisible;
    std::ofstream f(configPath);
    if (f.is_open()) {
        f << j.dump(2);
    }
}
```
