```javascript
// Helper: convert a blob to a base64 data URI
function blobToDataUrl(blob) {
    return new Promise(function(resolve, reject) {
        var reader = new FileReader();
        reader.onload = function() { resolve(reader.result); };
        reader.onerror = reject;
        reader.readAsDataURL(blob);
    });
}

// Full export implementation (D-09, EXPT-01)
async function exportHtml() {
    try {
        // Step 1: Collect CSS from all <link rel="stylesheet"> elements
        var links = Array.from(document.querySelectorAll('link[rel=stylesheet]'));
        var cssTexts = await Promise.all(
            links.map(function(link) {
                return fetch(link.href)
                    .then(function(r) { return r.text(); })
                    .catch(function() { return ''; });  // ignore fetch errors
            })
        );

        // Step 2: Collect custom CSS from inline <style id="custom-css"> if present
        var customStyle = document.getElementById('custom-css');
        var customCssText = customStyle ? customStyle.textContent : '';

        var inlinedCss = cssTexts.join('\n') + '\n' + customCssText;

        // Step 3: Clone preview div to avoid mutating live DOM
        var preview = document.getElementById('preview');
        var clone = preview.cloneNode(true);

        // Step 4: Convert local images to base64 data URIs in the clone
        var images = Array.from(clone.querySelectorAll('img'));
        await Promise.all(images.map(async function(img) {
            var originalSrc = img.getAttribute('src') || '';
            // Only process non-data-URI sources (data: URIs are already inline)
            if (originalSrc.startsWith('data:')) return;
            try {
                var resp = await fetch(originalSrc);
                if (!resp.ok) return;  // leave broken images as-is
                var blob = await resp.blob();
                img.src = await blobToDataUrl(blob);
            } catch (e) {
                // Remote image or unresolvable — leave src as original URL
            }
        }));

        // Step 5: Build standalone HTML document
        // Include current theme class on body for correct rendering
        var isDark = document.body.getAttribute('data-theme') === 'dark';
        var bodyClass = 'markdown-body';
        var html = '<!DOCTYPE html>\n'
            + '<html lang="en">\n'
            + '<head>\n'
            + '<meta charset="UTF-8">\n'
            + '<meta name="viewport" content="width=device-width, initial-scale=1.0">\n'
            + '<title>Exported Markdown</title>\n'
            + '<style>\n' + inlinedCss + '\n</style>\n'
            + '</head>\n'
            + '<body class="' + bodyClass + '"' + (isDark ? ' data-theme="dark"' : '') + '>\n'
            + clone.outerHTML + '\n'
            + '</body>\n'
            + '</html>';

        // Step 6: Send to C++ for file I/O (D-10)
        window.chrome.webview.postMessage(
            JSON.stringify({ type: 'exportReady', html: html })
        );
    } catch (err) {
        // Export failed — do not crash; C++ will receive no exportReady message
        // and nothing will be written (silent failure in Phase 2)
        console.error('Export failed:', err);
    }
}
```

```cpp
void PreviewPanel::triggerExport() {
    if (!m_webview || !m_webview2Initialized || m_currentFilePath.empty()) return;

    // Derive target path: replace .md extension with .html (D-10)
    std::wstring exportPath = m_currentFilePath;
    size_t dotPos = exportPath.rfind(L'.');
    if (dotPos != std::wstring::npos) {
        exportPath = exportPath.substr(0, dotPos) + L".html";
    } else {
        exportPath += L".html";
    }
    m_exportFilePath = exportPath;

    // Send export trigger to JS
    nlohmann::json j;
    j["type"] = "export";
    std::string jsonStr = j.dump();
    std::wstring wjson(jsonStr.begin(), jsonStr.end());
    m_webview->PostWebMessageAsJson(wjson.c_str());
}
```

```cpp
void PreviewPanel::saveExportedHtml(const std::string& htmlUtf8) {
    if (m_exportFilePath.empty()) return;

    // Write UTF-8 file (with BOM for maximum browser compatibility)
    // Silent overwrite per D-10 — no prompt
    std::ofstream f(m_exportFilePath, std::ios::out | std::ios::binary);
    if (!f.is_open()) return;

    // UTF-8 BOM: 0xEF, 0xBB, 0xBF
    const unsigned char bom[] = { 0xEF, 0xBB, 0xBF };
    f.write(reinterpret_cast<const char*>(bom), sizeof(bom));
    f.write(htmlUtf8.c_str(), static_cast<std::streamsize>(htmlUtf8.size()));
    f.close();
    m_exportFilePath.clear();  // reset after write
}
```

```cpp
void PreviewPanel::handleJsMessage(const std::wstring& message) {
    int utf8len = ::WideCharToMultiByte(CP_UTF8, 0, message.c_str(),
        static_cast<int>(message.size()), nullptr, 0, nullptr, nullptr);
    std::string utf8Msg(static_cast<size_t>(utf8len), '\0');
    ::WideCharToMultiByte(CP_UTF8, 0, message.c_str(),
        static_cast<int>(message.size()), &utf8Msg[0], utf8len, nullptr, nullptr);

    try {
        nlohmann::json j = nlohmann::json::parse(utf8Msg);
        std::string type = j.value("type", "");
        if (type == "exportReady") {
            // j["html"] is the full standalone HTML document as a UTF-8 string
            std::string htmlContent = j.value("html", "");
            if (!htmlContent.empty()) {
                saveExportedHtml(htmlContent);
            }
        }
    } catch (...) {
        // Malformed message — ignore silently
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| JS exportReady → C++ file write | JS sends full HTML document string back to C++ for file I/O. The string is the serialized DOM — no shell commands. |
| C++ → file system | saveExportedHtml writes to m_exportFilePath. The path is derived from m_currentFilePath (set by Notepad++ via NPPM_GETFULLCURRENTPATH). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-14 | Tampering | exportReady HTML payload | accept | The HTML content in exportReady comes from the JS DOM serialization of content the user typed. It cannot contain arbitrary C++ code. The C++ side writes the string to disk as-is — no execution, no shell invocation. Risk: malformed HTML in the output file. Impact: cosmetic only. |
| T-02-15 | Elevation of Privilege | saveExportedHtml path derivation | mitigate | Export path is derived by replacing .md with .html on m_currentFilePath. m_currentFilePath is set from NPPM_GETFULLCURRENTPATH — a Notepad++ API that returns only the path of the currently open file. The plugin cannot write to arbitrary paths via this flow. Path traversal via the filename is not possible since we only replace the extension, not accept an arbitrary path from JS. |
| T-02-16 | Denial of Service | Large HTML export payload | accept | Exporting a large document produces a large exportReady JSON string traversing the WebView2 IPC boundary. Practical maximum for markdown documents is well under 50MB. No mitigation needed for Phase 2. |
| T-02-17 | Information Disclosure | Base64 image encoding | accept | Images in the user's document are base64-encoded into the exported HTML. These are images already present in the file system adjacent to the .md file — the user intentionally embedded them. No new information disclosure. |
</threat_model>
