--- CHANGE 4: Add renderMermaidDiagrams() function ---
Add this function BEFORE the renderMarkdown() function:
```javascript
    // =========================================================================
    // Mermaid diagram rendering (XRND-02, D-01)
    // Called after hljs.highlightAll() in renderMarkdown().
    // hljs processes all code blocks including mermaid ones. We must:
    //   1. Find code elements with class "language-mermaid"
    //   2. Extract raw source (pre-highlighting text)
    //   3. Convert to <div class="mermaid"> for Mermaid processing
    //   4. Call mermaid.render() per diagram with try/catch for D-01 fallback
    // =========================================================================
    function renderMermaidDiagrams() {
        if (!_mermaid) return;

        var mermaidBlocks = document.querySelectorAll('code.language-mermaid');
        mermaidBlocks.forEach(function(codeEl, index) {
            var pre = codeEl.parentElement;
            // codeEl.textContent gives raw text (hljs adds <span> nodes but textContent strips them)
            var source = codeEl.textContent || '';

            // Create a unique ID for mermaid.render()
            var id = 'mermaid-diagram-' + index + '-' + Date.now();

            // D-01: Use mermaid.render() per diagram with try/catch.
            // mermaid.run() does not reject per-diagram errors — it inserts an error SVG.
            // mermaid.render() returns a Promise; we catch per diagram to implement D-01.
            _mermaid.render(id, source).then(function(result) {
                // result.svg is the rendered SVG string
                var container = document.createElement('div');
                container.className = 'mermaid-diagram';
                container.innerHTML = result.svg;
                pre.parentNode.replaceChild(container, pre);
            }).catch(function() {
                // D-01 fallback: show syntax-highlighted code block instead of error SVG
                // The <pre><code class="language-mermaid"> is already in the DOM from hljs;
                // just leave it in place (no DOM change needed on failure).
                // hljs has already highlighted it; it will display as code.
            });
        });
    }
```

--- CHANGE 5: Add injectCopyButtons() function ---
Add this function AFTER renderMermaidDiagrams() and BEFORE renderMarkdown():
```javascript
    // =========================================================================
    // Copy-to-clipboard button injection (XRND-04, D-02, D-03)
    // Called after renderMermaidDiagrams() in renderMarkdown().
    // Injects clipboard SVG button into each <pre><code> block.
    // Does NOT inject into mermaid diagram containers (they have no <code> child).
    // Silent copy per D-03 — no feedback animation.
    // navigator.clipboard.writeText() used (not deprecated execCommand).
    // =========================================================================
    function injectCopyButtons() {
        var blocks = document.querySelectorAll('pre > code');
        blocks.forEach(function(codeEl) {
            var pre = codeEl.parentElement;
            if (pre.querySelector('.copy-btn')) return;  // already injected (idempotent)

            var btn = document.createElement('button');
            btn.className = 'copy-btn';
            btn.setAttribute('aria-label', 'Copy code');
            btn.setAttribute('type', 'button');
            // GitHub clipboard SVG icon (D-02)
            btn.innerHTML = '<svg width="16" height="16" viewBox="0 0 16 16" fill="currentColor" aria-hidden="true">'
                + '<path d="M0 6.75C0 5.784.784 5 1.75 5h1.5a.75.75 0 0 1 0 1.5h-1.5a.25.25 0 0 0-.25.25v7.5c0 .138.112.25.25.25h7.5a.25.25 0 0 0 .25-.25v-1.5a.75.75 0 0 1 1.5 0v1.5A1.75 1.75 0 0 1 9.25 16h-7.5A1.75 1.75 0 0 1 0 14.25Z"/>'
                + '<path d="M5 1.75C5 .784 5.784 0 6.75 0h7.5C15.216 0 16 .784 16 1.75v7.5A1.75 1.75 0 0 1 14.25 11h-7.5A1.75 1.75 0 0 1 5 9.25Zm1.75-.25a.25.25 0 0 0-.25.25v7.5c0 .138.112.25.25.25h7.5a.25.25 0 0 0 .25-.25v-7.5a.25.25 0 0 0-.25-.25Z"/>'
                + '</svg>';

            // D-03: silent copy — no feedback, no checkmark swap, no label change
            btn.addEventListener('click', function() {
                var text = codeEl.textContent || '';
                if (navigator.clipboard && navigator.clipboard.writeText) {
                    navigator.clipboard.writeText(text).catch(function() {
                        // Clipboard API failed silently per D-03
                    });
                }
                // No fallback to execCommand — deprecated and unreliable
            });

            pre.appendChild(btn);
        });
    }
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| CDN → assets/ | Mermaid IIFE bundle downloaded from jsdelivr.net at build time; served locally |
| markdown content → Mermaid renderer | User-authored diagram source processed by Mermaid in WebView2 sandbox |
| copy button → Clipboard API | Code block text written to OS clipboard via navigator.clipboard |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-05 | Tampering | mermaid.min.js CDN download | mitigate | Pin exact version mermaid@11.14.0 in URL; verify file size > 2MB after download |
| T-03-06 | Injection | Mermaid diagram source (user markdown) | mitigate | Mermaid runs inside WebView2 sandbox; mermaid.render() does not execute arbitrary JS from diagram source; SVG output is injected as innerHTML but Mermaid sanitizes it; `html: false` on markdown-it prevents raw HTML passthrough |
| T-03-07 | Information Disclosure | Clipboard API | accept | navigator.clipboard.writeText() only writes user's own code block content to clipboard; user initiated action (click); no sensitive data pathway; WebView2 virtual host context grants clipboard access |
| T-03-08 | Denial of Service | Large Mermaid diagrams | accept | Mermaid runs async per diagram; a slow diagram does not block the preview; diagrams are user-authored so intentional slowness is user's own concern |
| T-03-09 | Spoofing | Mermaid error SVG injection on parse failure | mitigate | D-01 fallback: on mermaid.render() rejection, the existing `<pre><code>` is left in place (DOM unchanged on catch); no error SVG injected; no XSS vector from Mermaid error content |
</threat_model>
