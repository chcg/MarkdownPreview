CHANGE 3 — JS functions `buildToc()` and `updateTocActive()`:

Add the following block to the `<script>` section, AFTER the `scrollToLine()` function definition
(after its closing `}`) and BEFORE the export implementation comment:

```javascript
    // =========================================================================
    // Table of contents (SCRL-03, D-05, D-06, D-07, D-08, D-09)
    // buildToc(): generates TOC from rendered headings after each renderMarkdown().
    // updateTocActive(): highlights the nearest TOC entry on editor scroll.
    // =========================================================================
    function buildToc() {
        var toc = document.getElementById('toc');
        if (!toc) return;
        var headings = document.querySelectorAll(
            '#preview h1,#preview h2,#preview h3,#preview h4,#preview h5,#preview h6'
        );
        if (!headings.length) {
            // D-06: no headings — hide sidebar, let content take full width
            toc.style.display = 'none';
            return;
        }
        // D-05: headings exist — show sidebar
        toc.style.display = 'block';
        toc.innerHTML = '';
        headings.forEach(function(h) {
            var line = parseInt(h.getAttribute('data-line') || '-1', 10);
            var text = h.textContent || '';
            var a = document.createElement('a');
            a.className = 'toc-entry toc-' + h.tagName.toLowerCase();
            a.textContent = text;  // text-overflow:ellipsis handles long headings in CSS
            a.title = text;        // full text on tooltip hover
            a.href = '#';
            a.setAttribute('data-line', String(line));
            a.addEventListener('click', function(e) {
                e.preventDefault();
                e.stopPropagation();  // defense-in-depth: prevent bubbling
                // D-08: navigate preview (smooth scroll to heading element)
                h.scrollIntoView({ behavior: 'smooth', block: 'start' });
                // D-08: navigate editor (same lineClick mechanism as Plan 01 D-01/D-04)
                if (line >= 0) {
                    window.chrome.webview.postMessage(
                        JSON.stringify({ type: 'lineClick', line: line })
                    );
                }
            });
            toc.appendChild(a);
        });
    }

    function updateTocActive(line) {
        // D-09: highlight the TOC entry whose heading is nearest above the current scroll line.
        // 'line' is the editor's current caret/scroll line (0-indexed, from C++ scroll message).
        var entries = document.querySelectorAll('#toc .toc-entry');
        var best = null;
        entries.forEach(function(entry) {
            var entryLine = parseInt(entry.getAttribute('data-line') || '-1', 10);
            if (entryLine >= 0 && entryLine <= line) {
                best = entry;
            }
        });
        entries.forEach(function(e) { e.classList.remove('toc-active'); });
        if (best) { best.classList.add('toc-active'); }
    }
```
